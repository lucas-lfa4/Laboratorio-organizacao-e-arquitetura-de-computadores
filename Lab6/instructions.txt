lw x1,0(x0)      // x1 = endereco base do MMIO (0x400)
lw x5,12(x0)     // x5 = ONE (=1)
lw x6,16(x0)     // x6 = MASK_KEY3 (=0x8, isola KEY[3])
lw x7,20(x0)     // x7 = MASK_KEY2 (=0x4, isola KEY[2])
lw x8,24(x0)     // x8 = MASK_KEY1 (=0x2, isola KEY[1])
sub x4,x4,x4     // x4 = contador = 0
lw x3,4(x1)      // x3 = leitura inicial de KEY (estado "anterior")
lw x2,4(x1)      // LOOP: x2 = leitura atual de KEY (0x400+4 = 0x404)
and x9,x2,x6     // x9 = KEY3 agora (0 = pressionado)
beq x9,x0,8      // pressionado agora? -> checa estado anterior
beq x0,x0,16     // solto agora -> pula bloco do botao 3
and x10,x3,x6    // x10 = KEY3 antes (0 = ja estava pressionado)
beq x10,x0,8     // ja estava pressionado -> nao e borda nova, pula
add x4,x4,x5     // borda nova (solto->pressionado) -> incrementa
and x9,x2,x7     // x9 = KEY2 agora (0 = pressionado)
beq x9,x0,8	 // pressionado agora? -> checar estado anterior
beq x0,x0,16	 // pula 4 linhas
and x10,x3,x7    // x10 = KEY2 antes
beq x10,x0,8	 // KEY2 estava pressionado antes esta pressionado? se sim, pule 2 linhas
sub x4,x4,x5     // borda nova -> decrementa
and x9,x2,x8     // x9 = KEY1 agora
beq x9,x0,8	 // pressionado agora? -> checar estado anterior
beq x0,x0,16	 // pula 4 linhas
and x10,x3,x8    // x10 = KEY1 antes
beq x10,x0,8	 // pula 2 linhas
sub x4,x4,x4     // borda nova -> zera contador
sw x4,12(x1)     // LEDG <= contador[8:0]  (0x400+12 = 0x40C)
add x3,x2,x0     // prev = now (guarda leitura atual p/ proxima volta)
beq x0,x0,-84    // volta para LOOP
